class Solution {
    public int[] buildArray(int[] nums) {
        // create an 'ans' array with same length as nums
        int[] ans = new int[nums.length];
        // run a for loop from 0 to a ans.length - 1 and set ans[i] = nums[nums[i]] as mentioned in question
        for (int i = 0; i < ans.length; i++) {
            ans[i] = nums[nums[i]];
        }
        return ans;
    }
}
