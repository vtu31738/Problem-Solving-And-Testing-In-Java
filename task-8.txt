class Solution {
public int[] sortedSquares(int[] nums) {
int[] arr =new int[nums.length];
int i=0;
int j= nums.length -1;

    for(int k=nums.length -1;k>=0;k--){
        int camp1=nums[i]*nums[i];
        int camp2=nums[j]*nums[j];

        if(camp1>camp2){
            arr[k]=camp1;
            i++;
        }
        else{
            arr[k]=camp2;
            j--;
        }
    }
    return arr;
}
}
